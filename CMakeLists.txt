cmake_minimum_required(VERSION 3.22)
project(CyHWInfo VERSION 0.1.0 LANGUAGES CXX)

include(FetchContent)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)

option(CYHWINFO_BUILD_UI "Build the reusable CyUI hardware dashboard" OFF)
option(CYHWINFO_BUILD_APP "Build the Windows CyHWInfo monitoring application" OFF)

set(cyhwinfo_defaults Cpu Gpu GpuTelemetry GpuProcesses Memory Storage StorageHealth Motherboard Battery Network)
set(cyhwinfo_supported ${cyhwinfo_defaults} LibreHardwareMonitor)
set(CYHWINFO_COMPONENTS "${cyhwinfo_defaults}" CACHE STRING "Semicolon-separated CyHWInfo components to build")
set(cyhwinfo_components "${CYHWINFO_COMPONENTS}")
foreach(component IN LISTS cyhwinfo_components)
    if(NOT component IN_LIST cyhwinfo_supported)
        message(FATAL_ERROR "Unknown CyHWInfo component: ${component}")
    endif()
endforeach()
if("GpuTelemetry" IN_LIST cyhwinfo_components OR "GpuProcesses" IN_LIST cyhwinfo_components)
    list(APPEND cyhwinfo_components Gpu)
endif()
if("StorageHealth" IN_LIST cyhwinfo_components)
    list(APPEND cyhwinfo_components Storage)
endif()
list(REMOVE_DUPLICATES cyhwinfo_components)

set(cyhwinfo_hwinfo_components "${cyhwinfo_components}")
list(REMOVE_ITEM cyhwinfo_hwinfo_components LibreHardwareMonitor)
set(cyhwinfo_uses_hwinfo FALSE)
if(cyhwinfo_hwinfo_components)
    set(cyhwinfo_uses_hwinfo TRUE)
endif()

if("LibreHardwareMonitor" IN_LIST cyhwinfo_components AND NOT WIN32)
    message(FATAL_ERROR "The LibreHardwareMonitor component is available on Windows")
endif()

function(cyhwinfo_fetch_hwinfo)
    set(CMAKE_POSITION_INDEPENDENT_CODE ON)
    set(HWINFO_STATIC ON)
    set(HWINFO_SHARED OFF)
    set(BUILD_EXAMPLES OFF)
    set(BUILD_TESTING OFF)
    set(HWINFO_CMAKE_BINARY_DIR "${CMAKE_CURRENT_BINARY_DIR}/hwinfo")
    set(HWINFO_COMPUTER OFF)
    set(HWINFO_OS OFF)
    set(hwinfo_component_pairs CPU Cpu GPU Gpu RAM Memory DISK Storage MAINBOARD Motherboard BATTERY Battery NETWORK Network)
    while(hwinfo_component_pairs)
        list(POP_FRONT hwinfo_component_pairs upstream public)
        set(HWINFO_${upstream} OFF)
        if(public IN_LIST cyhwinfo_components)
            set(HWINFO_${upstream} ON)
        endif()
    endwhile()
    FetchContent_Declare(lfreist_hwinfo
        GIT_REPOSITORY https://github.com/lfreist/hwinfo.git
        GIT_TAG 1a77a8088919b98b417ab3ffd0ac5d0eb3a6ba96
        GIT_SUBMODULES ""
    )
    FetchContent_MakeAvailable(lfreist_hwinfo)
endfunction()

if(cyhwinfo_uses_hwinfo)
    cyhwinfo_fetch_hwinfo()
endif()

add_library(cyhwinfo_core INTERFACE)
add_library(CyHWInfo::Core ALIAS cyhwinfo_core)
set_target_properties(cyhwinfo_core PROPERTIES EXPORT_NAME Core)
target_compile_features(cyhwinfo_core INTERFACE cxx_std_23)
target_include_directories(cyhwinfo_core INTERFACE
    "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>"
    "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>"
)

add_library(cyhwinfo_all INTERFACE)
add_library(CyHWInfo::All ALIAS cyhwinfo_all)
set_target_properties(cyhwinfo_all PROPERTIES EXPORT_NAME All)
target_link_libraries(cyhwinfo_all INTERFACE cyhwinfo_core)

function(cyhwinfo_component name file)
    if(NOT name IN_LIST cyhwinfo_components)
        return()
    endif()
    set(target cyhwinfo_${file})
    add_library(${target} STATIC src/${file}.cpp)
    add_library(CyHWInfo::${name} ALIAS ${target})
    set_target_properties(${target} PROPERTIES EXPORT_NAME "${name}" CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
    target_link_libraries(${target} PUBLIC cyhwinfo_core)
    if(WIN32)
        target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    endif()
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    target_link_libraries(cyhwinfo_all INTERFACE ${target})
    install(TARGETS ${target} EXPORT CyHWInfoTargets ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}")
    install(FILES "include/cyhwinfo/${file}.hpp" DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/cyhwinfo")
endfunction()

cyhwinfo_component(Cpu cpu)
cyhwinfo_component(Gpu gpu)
cyhwinfo_component(GpuTelemetry gpu_telemetry)
cyhwinfo_component(GpuProcesses gpu_processes)
cyhwinfo_component(Memory memory)
cyhwinfo_component(Storage storage)
cyhwinfo_component(StorageHealth storage_health)
cyhwinfo_component(Motherboard motherboard)
cyhwinfo_component(Battery battery)
cyhwinfo_component(Network network)
cyhwinfo_component(LibreHardwareMonitor sensors)

foreach(pair IN ITEMS cpu:cpu gpu:gpu memory:ram storage:disk motherboard:mainboard battery:battery network:network)
    string(REPLACE ":" ";" split "${pair}")
    list(GET split 0 component)
    list(GET split 1 upstream)
    if(TARGET cyhwinfo_${component})
        target_link_libraries(cyhwinfo_${component} PRIVATE lfreist-hwinfo::${upstream})
    endif()
endforeach()

if(TARGET cyhwinfo_gpu_telemetry)
    target_link_libraries(cyhwinfo_gpu_telemetry PUBLIC cyhwinfo_gpu PRIVATE lfreist-hwinfo::gpu)
    if(WIN32)
        target_link_libraries(cyhwinfo_gpu_telemetry PRIVATE gdi32)
    endif()
endif()
if(TARGET cyhwinfo_gpu_processes)
    target_link_libraries(cyhwinfo_gpu_processes PUBLIC cyhwinfo_gpu)
    if(WIN32)
        target_link_libraries(cyhwinfo_gpu_processes PRIVATE pdh)
    endif()
endif()
if(TARGET cyhwinfo_storage_health)
    target_link_libraries(cyhwinfo_storage_health PUBLIC cyhwinfo_storage)
endif()
if(TARGET cyhwinfo_sensors)
    include(cmake/LibreHardwareMonitor.cmake)
endif()
if(CYHWINFO_BUILD_UI OR CYHWINFO_BUILD_APP)
    include(cmake/CyHWInfoUI.cmake)
endif()

set(cyhwinfo_package_dir "${CMAKE_INSTALL_LIBDIR}/cmake/CyHWInfo")
configure_package_config_file(cmake/CyHWInfoConfig.cmake.in
    "${CMAKE_CURRENT_BINARY_DIR}/CyHWInfoConfig.cmake"
    INSTALL_DESTINATION "${cyhwinfo_package_dir}"
    PATH_VARS CMAKE_INSTALL_BINDIR
)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/CyHWInfoConfigVersion.cmake"
    VERSION "${PROJECT_VERSION}" COMPATIBILITY SameMinorVersion
)
install(TARGETS cyhwinfo_core cyhwinfo_all EXPORT CyHWInfoTargets)
install(EXPORT CyHWInfoTargets NAMESPACE CyHWInfo:: DESTINATION "${cyhwinfo_package_dir}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/CyHWInfoConfig.cmake"
              "${CMAKE_CURRENT_BINARY_DIR}/CyHWInfoConfigVersion.cmake"
        DESTINATION "${cyhwinfo_package_dir}")
install(FILES include/cyhwinfo/core.hpp DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/cyhwinfo")
install(FILES LICENSE THIRD-PARTY-NOTICES.txt DESTINATION "${CMAKE_INSTALL_DATADIR}/CyHWInfo")
